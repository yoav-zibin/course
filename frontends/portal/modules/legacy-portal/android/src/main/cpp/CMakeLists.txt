cmake_minimum_required(VERSION 3.18.1)
project(buildplay-storage LANGUAGES CXX)
find_package(ReactAndroid REQUIRED CONFIG)
find_package(fbjni REQUIRED CONFIG)
add_library(buildplay-storage SHARED StorageWorklet.cpp)
target_compile_features(buildplay-storage PRIVATE cxx_std_20)
target_include_directories(buildplay-storage PRIVATE ../../../../cpp)
target_link_libraries(buildplay-storage ReactAndroid::jsi fbjni::fbjni)
target_link_options(buildplay-storage PRIVATE "-Wl,-z,max-page-size=16384")
